cmake_minimum_required(VERSION 3.21)
project(JatayuLekh VERSION 0.2.6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets PrintSupport)
set(LEKH_REPOSITORY_URL "https://github.com/jatayuos4u/jatayu-lekh" CACHE STRING "Verified GitHub repository URL for Lekh")
set(LEKH_LICENSE_URL "https://github.com/jatayuos4u/jatayu-lekh/blob/main/LICENSE" CACHE STRING "Verified GitHub URL for Lekh's license")
foreach(link IN ITEMS LEKH_REPOSITORY_URL LEKH_LICENSE_URL)
    if(NOT "${${link}}" STREQUAL "" AND NOT "${${link}}" MATCHES "^https://github[.]com/[A-Za-z0-9._/-]+$")
        message(FATAL_ERROR "${link} must be an HTTPS GitHub URL")
    endif()
endforeach()
configure_file(projectlinks.h.in projectlinks.h @ONLY)
add_executable(jatayu-lekh main.cpp assets/resources.qrc)
target_include_directories(jatayu-lekh PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(jatayu-lekh PRIVATE Qt6::Widgets Qt6::PrintSupport)
install(TARGETS jatayu-lekh RUNTIME DESTINATION bin)
install(FILES LICENSE DESTINATION share/doc/jatayu-lekh)
install(FILES assets/in.jatayuos.Lekh.desktop DESTINATION share/applications)
install(FILES assets/in.jatayuos.Lekh.svg DESTINATION share/icons/hicolor/scalable/apps)
include(CTest)
if(BUILD_TESTING)
    add_test(NAME editor-smoke COMMAND jatayu-lekh --smoke-test)
    set_tests_properties(editor-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME jatayu-lekh)
set(CPACK_PACKAGE_CONTACT "JatayuOS <https://jatayuos.in>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Dual-mode desktop text editor for JatayuOS")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_SECTION editors)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6svg6, diffutils, libhunspell-1.7-0, hunspell-en-us")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
include(CPack)
